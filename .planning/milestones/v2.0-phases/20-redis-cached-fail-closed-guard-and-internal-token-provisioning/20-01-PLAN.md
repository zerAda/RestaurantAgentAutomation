  buildCacheKey(tenant_id, module_key) -> string
      // EXACTLY `ralphe:entitlement:${tenant_id}:${module_key}` — byte-for-byte == audit-hook.ts:122 / ADR 0003.

  decideFromCache(cachedRaw, now) -> {
      cacheUsable: boolean,   // true ONLY when cachedRaw is a parseable {ent,mod,fetchedAt}|{neg:true} row
      allowed?:   boolean,    // present iff cacheUsable
      reason?:    string,     // ENTITLED_CACHED | GLOBAL_ENABLED_CACHED | EXPIRED | NO_ENTITLEMENT
      config_overrides?: object
    }
      // cachedRaw === null / '' / 'nil' / a {error} object (Redis continueOnFail envelope) → cacheUsable:false (MISS/fall-through).
      // a parseable {neg:true} sentinel → cacheUsable:true, allowed:false, reason 'NO_ENTITLEMENT'.
      // a parseable {ent,mod,fetchedAt} → RE-EVALUATE ent.expires_at vs `now`: past → allowed:false reason 'EXPIRED';
      //   future/absent → allowed:true reason 'ENTITLED_CACHED' (or 'GLOBAL_ENABLED_CACHED' if mod.tier==='shared_core'||mod.enabled_globally).
      // NEVER returns a transient error as a usable cache hit. A miss is NEVER a deny (LRU eviction = miss → live).

  evaluateLive(moduleRes, entRes, now) -> {
      allowed:  boolean,
      reason:   string,       // MODULE_NOT_FOUND | GLOBAL_ENABLED | NO_ENTITLEMENT | EXPIRED | ENTITLED | GUARD_ERROR_FAILCLOSED: <detail>
      config_overrides: object,
      cacheRow?: object,      // {ent,mod,fetchedAt} on a positive/entitled result; {neg:true} on a definitive negative
      cacheable: boolean,     // true for ENTITLED/GLOBAL_ENABLED/NO_ENTITLEMENT/MODULE_NOT_FOUND/EXPIRED; FALSE for GUARD_ERROR_FAILCLOSED
      ttl: number             // positive TTL for positive rows; negative TTL for NO_ENTITLEMENT/MODULE_NOT_FOUND; 0 when !cacheable
    }
      // moduleRes/entRes are the Strapi responses (or a {error}/non-2xx/empty-body envelope from httpRequest continueOnFail).
      // Strapi error detection (Pitfall 8): $json.error present OR non-2xx OR structurally-invalid body → GUARD_ERROR_FAILCLOSED
      //   (NOT NO_ENTITLEMENT — a missing token must page, not masquerade as a routine denial). Distinguish
      //   "valid response, zero rows" (NO_ENTITLEMENT/MODULE_NOT_FOUND) from "couldn't get a valid response" (FAILCLOSED).
      // TTLs are READ from the caller: evaluateLive(moduleRes, entRes, now, { posTtl, negTtl }) — default posTtl=300, negTtl=60.
