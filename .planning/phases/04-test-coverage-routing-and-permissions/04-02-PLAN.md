Create `scripts/smoke-strapi-permissions.sh`. This script validates the Strapi permission matrix for both the Public role (unauthenticated kiosk access) and Authenticated role (admin dashboard access).

The script takes the following env vars (all optional with defaults):
- `CMS_URL` — Strapi base URL, default `http://127.0.0.1:1337`
- `STRAPI_EMAIL` — user email for Authenticated role test (users-permissions user, NOT admin panel)
- `STRAPI_PASSWORD` — user password for Authenticated role test
- `STRAPI_ADMIN_EMAIL` — Strapi admin panel email for seeding (optional, used only in seed mode)
- `STRAPI_ADMIN_PASSWORD` — Strapi admin panel password for seeding (optional)
- `SEED_PERMISSIONS` — if set to "true", grant Public role permissions before testing

It also accepts positional args: `./smoke-strapi-permissions.sh [CMS_URL] [EMAIL] [PASSWORD]`

Write the complete script:

```bash
#!/usr/bin/env bash
# smoke-strapi-permissions.sh — Strapi role permission matrix smoke test
# Tests: TEST-05 (Public GET /api/products → 200), TEST-06 (unauth POST /api/orders → 403/401),
#        TEST-07 (Authenticated GET /api/orders → 200)
#
# Usage:
#   bash scripts/smoke-strapi-permissions.sh [CMS_URL] [EMAIL] [PASSWORD]
#   CMS_URL=http://localhost:1337 STRAPI_EMAIL=user@example.com STRAPI_PASSWORD=pass \
#     bash scripts/smoke-strapi-permissions.sh
#
# Exit codes:
#   0 — all permission checks passed
#   1 — one or more permission checks failed
#   2 — infrastructure error (Strapi unreachable, JWT not obtained)

set -euo pipefail

CMS_URL="${1:-${CMS_URL:-http://127.0.0.1:1337}}"
EMAIL="${2:-${STRAPI_EMAIL:-}}"
PASS="${3:-${STRAPI_PASSWORD:-}}"
SEED_PERMISSIONS="${SEED_PERMISSIONS:-false}"
STRAPI_ADMIN_EMAIL="${STRAPI_ADMIN_EMAIL:-}"
STRAPI_ADMIN_PASSWORD="${STRAPI_ADMIN_PASSWORD:-}"

PASS_COUNT=0
FAIL_COUNT=0
ERRORS=()

pass() { echo "PASS  $1"; PASS_COUNT=$((PASS_COUNT + 1)); }
fail() { echo "FAIL  $1  — $2"; FAIL_COUNT=$((FAIL_COUNT + 1)); ERRORS+=("$1: $2"); }

echo "=== Strapi Permission Matrix Smoke Test ==="
echo "CMS_URL: ${CMS_URL}"
echo ""

# ============================================================
# Preflight: verify Strapi is reachable
# ============================================================
echo "--- Preflight: Strapi health check ---"
HEALTH_STATUS=$(curl -s -o /dev/null -w "%{http_code}" --connect-timeout 5 "${CMS_URL}/_health" 2>/dev/null || echo "000")
if [ "${HEALTH_STATUS}" = "204" ] || [ "${HEALTH_STATUS}" = "200" ]; then
  echo "Strapi healthy (${HEALTH_STATUS})"
else
  echo "ERROR: Strapi not reachable at ${CMS_URL} — health returned ${HEALTH_STATUS}"
  echo "Ensure CMS is running: docker exec current-cms-1 curl -s http://localhost:1337/_health"
  exit 2
fi
echo ""

# ============================================================
# Optional: seed Public role permissions via admin API
# Set SEED_PERMISSIONS=true + STRAPI_ADMIN_EMAIL + STRAPI_ADMIN_PASSWORD to activate
# ============================================================
if [ "${SEED_PERMISSIONS}" = "true" ]; then
  echo "--- Seeding Public role permissions (SEED_PERMISSIONS=true) ---"
  if [ -z "${STRAPI_ADMIN_EMAIL}" ] || [ -z "${STRAPI_ADMIN_PASSWORD}" ]; then
    echo "WARNING: STRAPI_ADMIN_EMAIL/PASSWORD not set — skipping seed"
  else
    # Get Strapi admin JWT (admin panel login endpoint)
    ADMIN_AUTH=$(curl -s -X POST "${CMS_URL}/admin/login" \
      -H 'Content-Type: application/json' \
      -d "{\"email\":\"${STRAPI_ADMIN_EMAIL}\",\"password\":\"${STRAPI_ADMIN_PASSWORD}\"}")
    if command -v jq >/dev/null 2>&1; then
      ADMIN_JWT=$(echo "${ADMIN_AUTH}" | jq -r '.data.token // empty')
    else
      ADMIN_JWT=$(echo "${ADMIN_AUTH}" | python3 -c "import sys,json; d=json.load(sys.stdin); print(d.get('data',{}).get('token',''))" 2>/dev/null || echo "")
    fi

    if [ -n "${ADMIN_JWT}" ]; then
      # Grant product.find and product.findOne to Public role (roleId=2)
      # Strapi 5 admin API: PUT /admin/users-permissions/roles/{id}/permissions
      SEED_RESULT=$(curl -s -o /dev/null -w "%{http_code}" \
        -X PUT "${CMS_URL}/admin/users-permissions/roles/2/permissions" \
        -H "Authorization: Bearer ${ADMIN_JWT}" \
        -H 'Content-Type: application/json' \
        -d '{
          "permissions": {
            "api::product.product": {
              "controllers": {
                "product": {
                  "find": {"enabled": true},
                  "findOne": {"enabled": true}
                }
              }
            }
          }
        }')
      echo "Seed Public role permissions: ${SEED_RESULT}"
    else
      echo "WARNING: Could not get admin JWT — seeding skipped"
    fi
  fi
  echo ""
fi

# ============================================================
# TEST-05: Unauthenticated GET /api/products → 200 with data
# No Authorization header = treated as Public role
# Public role has product.find permission (seeded or existing)
# ============================================================
echo "--- TEST-05: Unauthenticated GET /api/products ---"
PRODUCTS_BODY=$(mktemp)
PROD_STATUS=$(curl -s -o "${PRODUCTS_BODY}" -w "%{http_code}" \
  --connect-timeout 10 \
  "${CMS_URL}/api/products")

if [ "${PROD_STATUS}" = "200" ]; then
  pass "GET /api/products unauth → 200"
else
  fail "GET /api/products unauth" "got ${PROD_STATUS}, expected 200 (check Public role has product.find permission)"
fi

# Verify response has data array (not an error body)
if grep -q '"data"' "${PRODUCTS_BODY}" 2>/dev/null; then
  pass "GET /api/products body contains 'data' key"
else
  BODY_PREVIEW=$(head -c 200 "${PRODUCTS_BODY}" 2>/dev/null || echo "(empty)")
  fail "GET /api/products body" "no 'data' key — got: ${BODY_PREVIEW}"
fi
rm -f "${PRODUCTS_BODY}"
echo ""

# ============================================================
# TEST-06: Unauthenticated POST /api/orders → 403 or 401
# No Authorization header = Public role
# Public role has NO order create permission
# Strapi 5: no token + no permission = 403 Forbidden
# ============================================================
echo "--- TEST-06: Unauthenticated POST /api/orders ---"
ORDER_STATUS=$(curl -s -o /dev/null -w "%{http_code}" \
  --connect-timeout 10 \
  -X POST "${CMS_URL}/api/orders" \
  -H "Content-Type: application/json" \
  -d '{"data":{"status":"new"}}')

if [ "${ORDER_STATUS}" = "403" ] || [ "${ORDER_STATUS}" = "401" ]; then
  pass "POST /api/orders unauth → ${ORDER_STATUS} (access correctly denied)"
else
  fail "POST /api/orders unauth" "got ${ORDER_STATUS}, expected 403 or 401 (Public role must NOT create orders)"
fi
echo ""

# ============================================================
# TEST-07: Authenticated GET /api/orders → 200 with data
# Requires STRAPI_EMAIL + STRAPI_PASSWORD (users-permissions user, NOT admin panel user)
# Authenticated role has full CRUD on orders
# ============================================================
echo "--- TEST-07: Authenticated GET /api/orders ---"
if [ -z "${EMAIL}" ] || [ -z "${PASS}" ]; then
  echo "SKIP: STRAPI_EMAIL and STRAPI_PASSWORD not set — skipping TEST-07 (authenticated test)"
  echo "      Set env vars to test: STRAPI_EMAIL=adel.zeriri@gmail.com STRAPI_PASSWORD=RestoBot2026"
else
  # Obtain JWT via users-permissions endpoint (NOT admin login)
  AUTH_RESPONSE=$(curl -s -X POST "${CMS_URL}/api/auth/local" \
    -H 'Content-Type: application/json' \
    -d "{\"identifier\":\"${EMAIL}\",\"password\":\"${PASS}\"}")

  if command -v jq >/dev/null 2>&1; then
    TOKEN=$(echo "${AUTH_RESPONSE}" | jq -r '.jwt // empty')
  else
    TOKEN=$(echo "${AUTH_RESPONSE}" | python3 -c "import sys,json; d=json.load(sys.stdin); print(d.get('jwt',''))" 2>/dev/null || echo "")
  fi

  if [ -z "${TOKEN}" ]; then
    echo "ERROR: Could not obtain JWT for ${EMAIL}"
    echo "Response: ${AUTH_RESPONSE}"
    exit 2
  fi
  echo "JWT obtained for ${EMAIL}"

  ORDERS_BODY=$(mktemp)
  ORDERS_STATUS=$(curl -s -o "${ORDERS_BODY}" -w "%{http_code}" \
    --connect-timeout 10 \
    -H "Authorization: Bearer ${TOKEN}" \
    "${CMS_URL}/api/orders")

  if [ "${ORDERS_STATUS}" = "200" ]; then
    pass "GET /api/orders authenticated → 200"
  else
    fail "GET /api/orders authenticated" "got ${ORDERS_STATUS}, expected 200 (check Authenticated role has order.find permission)"
  fi

  # Verify response has data array
  if grep -q '"data"' "${ORDERS_BODY}" 2>/dev/null; then
    pass "GET /api/orders body contains 'data' key"
  else
    BODY_PREVIEW=$(head -c 200 "${ORDERS_BODY}" 2>/dev/null || echo "(empty)")
    fail "GET /api/orders body" "no 'data' key — got: ${BODY_PREVIEW}"
  fi
  rm -f "${ORDERS_BODY}"
fi
echo ""

# ============================================================
# Summary
# ============================================================
TOTAL=$((PASS_COUNT + FAIL_COUNT))
echo "=== Results: ${PASS_COUNT}/${TOTAL} passed ==="

if [ "${FAIL_COUNT}" -gt 0 ]; then
  echo ""
  echo "FAILURES:"
  for err in "${ERRORS[@]}"; do
    echo "  - ${err}"
  done
  echo ""
  echo "Troubleshooting:"
  echo "  TEST-05 fails 403: Public role missing product.find — run with SEED_PERMISSIONS=true"
  echo "  TEST-06 fails 200: Public role incorrectly has order.create — remove from up_permissions"
  echo "  TEST-07 fails 401: JWT invalid — check STRAPI_EMAIL/STRAPI_PASSWORD"
  echo "  TEST-07 fails 403: Authenticated role missing order.find — grant via Strapi admin UI"
  exit 1
fi

echo "All permission checks OK"
exit 0
```
