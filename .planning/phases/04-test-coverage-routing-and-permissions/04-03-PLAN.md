**Edit 2: Add the `strapi-permissions` job** — insert immediately after the `nginx-smoke` job block (still before `ci-summary`). Add this complete job block:

```yaml
  # ==========================================================================
  # JOB 11: Strapi Permission Matrix (main/release + inventory-cms PRs)
  # Tests: TEST-05 (public GET /api/products), TEST-06 (unauth POST /api/orders),
  #        TEST-07 (authenticated GET /api/orders)
  # Timeout: 15 min — Strapi first-boot takes 2-3 min in CI
  # ==========================================================================
  strapi-permissions:
    name: Strapi Permission Matrix
    runs-on: ubuntu-latest
    needs: [integrity-gate]
    if: |
      github.ref == 'refs/heads/main' ||
      startsWith(github.ref, 'refs/heads/release/') ||
      contains(github.event.pull_request.title, '[cms]') ||
      (github.event_name == 'pull_request' && (
        startsWith(github.event.pull_request.head.ref, 'cms/') ||
        startsWith(github.event.pull_request.head.ref, 'feat/cms')
      ))
    timeout-minutes: 15

    services:
      postgres:
        image: postgres:15-alpine
        env:
          POSTGRES_USER: strapi
          POSTGRES_PASSWORD: strapi
          POSTGRES_DB: strapi
        ports:
          - 5432:5432
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

    steps:
      - name: Checkout code
        uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4.2.2

      - name: Log in to GHCR
        run: echo "${{ secrets.GITHUB_TOKEN }}" | docker login ghcr.io -u ${{ github.actor }} --password-stdin

      - name: Pull CMS image
        run: docker pull ghcr.io/${{ github.repository_owner }}/resto-bot-cms:latest

      - name: Start CMS container
        env:
          APP_KEYS: ${{ secrets.STRAPI_APP_KEYS || 'ci-key-1,ci-key-2,ci-key-3,ci-key-4' }}
          ADMIN_JWT_SECRET: ${{ secrets.STRAPI_ADMIN_JWT_SECRET || 'ci-admin-jwt-secret-32chars-xx' }}
          JWT_SECRET: ${{ secrets.STRAPI_JWT_SECRET || 'ci-jwt-secret-32-characters-xxx' }}
          API_TOKEN_SALT: ${{ secrets.STRAPI_API_TOKEN_SALT || 'ci-api-token-salt-value-here-xx' }}
        run: |
          docker run -d --name strapi-ci \
            --network host \
            -e NODE_ENV=production \
            -e APP_KEYS="${APP_KEYS}" \
            -e ADMIN_JWT_SECRET="${ADMIN_JWT_SECRET}" \
            -e JWT_SECRET="${JWT_SECRET}" \
            -e API_TOKEN_SALT="${API_TOKEN_SALT}" \
            -e DATABASE_CLIENT=postgres \
            -e DATABASE_HOST=127.0.0.1 \
            -e DATABASE_PORT=5432 \
            -e DATABASE_NAME=strapi \
            -e DATABASE_USERNAME=strapi \
            -e DATABASE_PASSWORD=strapi \
            -e DATABASE_SSL=false \
            -p 1337:1337 \
            ghcr.io/${{ github.repository_owner }}/resto-bot-cms:latest

      - name: Wait for Strapi to be ready (max 8 min)
        run: |
          echo "Waiting for Strapi /_health..."
          WAITED=0
          until curl -s -o /dev/null -w "%{http_code}" http://127.0.0.1:1337/_health | grep -qE "^(200|204)$"; do
            if [ "${WAITED}" -ge 480 ]; then
              echo "::error::Strapi did not become healthy within 8 minutes"
              docker logs strapi-ci --tail 50 || true
              exit 1
            fi
            sleep 5
            WAITED=$((WAITED + 5))
            echo "  waited ${WAITED}s..."
          done
          echo "Strapi ready after ${WAITED}s"

      - name: Seed Public role permissions
        run: |
          # Strapi 5: Public role (id=2) needs product.find + product.findOne
          # Use admin API to seed — admin user credentials from env or use DB direct
          ADMIN_AUTH=$(curl -s -X POST http://127.0.0.1:1337/admin/login \
            -H 'Content-Type: application/json' \
            -d "{\"email\":\"${{ secrets.STRAPI_ADMIN_EMAIL || 'adel.zeriri@gmail.com' }}\",\"password\":\"${{ secrets.STRAPI_ADMIN_PASSWORD || 'RestoBot2026' }}\"}" \
            2>/dev/null || echo "{}")
          ADMIN_JWT=$(echo "${ADMIN_AUTH}" | python3 -c "import sys,json; d=json.load(sys.stdin); print(d.get('data',{}).get('token',''))" 2>/dev/null || echo "")

          if [ -n "${ADMIN_JWT}" ]; then
            # Get current Public role permissions and update with product.find + product.findOne
            echo "Seeding Public role permissions via admin API..."
            HTTP_RESULT=$(curl -s -o /dev/null -w "%{http_code}" \
              -X PUT http://127.0.0.1:1337/admin/users-permissions/roles/2/permissions \
              -H "Authorization: Bearer ${ADMIN_JWT}" \
              -H 'Content-Type: application/json' \
              -d '{"permissions":{"api::product.product":{"controllers":{"product":{"find":{"enabled":true},"findOne":{"enabled":true}}}}}}')
            echo "Seed result: ${HTTP_RESULT}"
          else
            echo "::warning::Admin JWT not obtained — Public role may not have product.find permission"
            echo "Admin auth response: ${ADMIN_AUTH}" | head -c 200
          fi

      - name: Run Strapi permission smoke test
        env:
          CMS_URL: http://127.0.0.1:1337
          STRAPI_EMAIL: ${{ secrets.STRAPI_EMAIL || 'adel.zeriri@gmail.com' }}
          STRAPI_PASSWORD: ${{ secrets.STRAPI_PASSWORD || 'RestoBot2026' }}
        run: bash scripts/smoke-strapi-permissions.sh

      - name: Upload Strapi logs on failure
        if: failure()
        run: docker logs strapi-ci --tail 100 2>&1 || true

      - name: Stop CMS container
        if: always()
        run: docker rm -f strapi-ci 2>/dev/null || true
```
