The workflow structure:

**Node 1 — Schedule Trigger (every 5 minutes):**
```json
{
  "parameters": {
    "rule": {
      "interval": [{ "field": "minutes", "minutesInterval": 5 }]
    }
  },
  "id": "queue-metrics-cron",
  "name": "CRON - Every 5min",
  "type": "n8n-nodes-base.scheduleTrigger",
  "typeVersion": 1,
  "position": [-800, 0]
}
```

**Node 2 — Code: Read thresholds from env (no DB access):**
```json
{
  "parameters": {
    "language": "javascript",
    "jsCode": "const queueThreshold = parseInt($env.QUEUE_ALERT_THRESHOLD || '50', 10);\nconst diskAlertPct = parseInt($env.DISK_ALERT_PCT || '80', 10);\nconst diskTotalGB = 119;\nreturn [{ json: { queueThreshold, diskAlertPct, diskTotalGB, ts: new Date().toISOString() } }];"
  },
  "id": "metrics-config",
  "name": "B0 - Config",
  "type": "n8n-nodes-base.code",
  "typeVersion": 2,
  "position": [-550, 0]
}
```

**Node 3 — Postgres: Query execution_entity for queue depth and error rate:**
```json
{
  "parameters": {
    "operation": "executeQuery",
    "query": "SELECT\n  COUNT(*) FILTER (WHERE status IN ('new','waiting')) AS pending_count,\n  COUNT(*) FILTER (WHERE status = 'running') AS running_count,\n  COUNT(*) FILTER (WHERE status IN ('error','crashed') AND \"startedAt\" > now() - interval '1 hour') AS errors_last_hour,\n  COUNT(*) FILTER (WHERE \"startedAt\" > now() - interval '1 hour') AS total_last_hour\nFROM execution_entity;",
    "options": {}
  },
  "credentials": {
    "postgres": {
      "id": "CREDENTIAL_ID_PLACEHOLDER",
      "name": "RestoBot PG (n8n DB)"
    }
  },
  "id": "query-queue-depth",
  "name": "PG - Queue Depth",
  "type": "n8n-nodes-base.postgres",
  "typeVersion": 2,
  "position": [-300, 0]
}
```

**Node 4 — Code: Compute metrics, check disk, emit structured log lines:**
```javascript
// This code runs inside n8n (Node.js). execSync is available for disk check.
const { execSync } = require('child_process');

const cfg = $('B0 - Config').first().json;
const pgRow = $input.first().json;

const pendingCount = parseInt(pgRow.pending_count || '0', 10);
const runningCount = parseInt(pgRow.running_count || '0', 10);
const errorsLastHour = parseInt(pgRow.errors_last_hour || '0', 10);
const totalLastHour = parseInt(pgRow.total_last_hour || '0', 10);
const errorRate = totalLastHour > 0 ? (errorsLastHour / totalLastHour) : 0;

// Disk check via df
let diskUsedGB = 0;
let diskUsedPct = 0;
try {
  const dfOut = execSync('df -k /').toString();
  const lines = dfOut.trim().split('\n');
  if (lines.length >= 2) {
    const parts = lines[1].trim().split(/\s+/);
    const usedKB = parseInt(parts[2], 10);
    diskUsedGB = Math.round(usedKB / 1024 / 1024 * 10) / 10;
    diskUsedPct = parseInt(parts[4].replace('%', ''), 10);
  }
} catch (e) {
  diskUsedPct = -1;
}

const ts = new Date().toISOString();

// Emit queue depth metric as structured JSON log (METR-01)
console.log(JSON.stringify({
  level: 'INFO',
  event: 'QUEUE_DEPTH_METRIC',
  pending_count: pendingCount,
  running_count: runningCount,
  errors_last_hour: errorsLastHour,
  error_rate_1h: Math.round(errorRate * 1000) / 1000,
  ts
}));

// Emit disk metric (METR-04)
console.log(JSON.stringify({
  level: 'INFO',
  event: 'DISK_USAGE_METRIC',
  disk_used_gb: diskUsedGB,
  disk_used_pct: diskUsedPct,
  disk_total_gb: cfg.diskTotalGB,
  ts
}));

const alerts = [];

// Queue depth alert: uses Redis counter for sustained detection
// Key: ralphe:metrics:queue_above_threshold_count
// If pending > threshold this window, increment; if >= 2 consecutive windows, fire CRITICAL
// We set the counter in next node (Redis node). Here we just flag.
const queueAbove = pendingCount > cfg.queueThreshold;

// Disk alert: immediate (single window) (METR-04)
if (diskUsedPct >= cfg.diskAlertPct) {
  const msg = {
    level: 'CRITICAL',
    event: 'DISK_ALERT',
    message: `Disk usage ${diskUsedPct}% exceeds threshold ${cfg.diskAlertPct}%`,
    disk_used_gb: diskUsedGB,
    disk_used_pct: diskUsedPct,
    disk_total_gb: cfg.diskTotalGB,
    ts
  };
  console.log(JSON.stringify(msg));
  alerts.push(msg);
}

return [{ json: {
  pendingCount, runningCount, errorsLastHour, errorRate,
  diskUsedGB, diskUsedPct,
  queueAbove,
  queueThreshold: cfg.queueThreshold,
  diskAlertPct: cfg.diskAlertPct,
  alerts,
  ts
}}];
```

**Node 6 — Code: Fire CRITICAL if sustained, then SET or RESET Redis counter:**
```javascript
const metrics = $('B0 - Compute Metrics').first().json;
const redisVal = $input.first().json;
const currentCount = parseInt(redisVal?.value || redisVal || '0', 10) || 0;

const { execSync } = require('child_process');
// Note: Redis SET is done via the next Redis SET node based on output.
// Here we decide: if queue above threshold, newCount = currentCount + 1; else reset to 0.
const newCount = metrics.queueAbove ? currentCount + 1 : 0;

let queueAlert = null;
if (metrics.queueAbove && newCount >= 2) {
  queueAlert = {
    level: 'CRITICAL',
    event: 'QUEUE_DEPTH_ALERT',
    message: `Queue depth ${metrics.pendingCount} exceeds threshold ${metrics.queueThreshold} for ${newCount} consecutive 5-min windows`,
    pending_count: metrics.pendingCount,
    threshold: metrics.queueThreshold,
    consecutive_windows: newCount,
    ts: metrics.ts
  };
  console.log(JSON.stringify(queueAlert));
}

return [{ json: { ...metrics, newCount, queueAlert } }];
```

**Node 7 — Redis SET: Update sustained queue alert counter (with 15-min TTL):**
```json
{
  "parameters": {
    "operation": "set",
    "key": "ralphe:metrics:queue_above_threshold_count",
    "value": "={{$json.newCount}}",
    "expire": true,
    "ttl": 900
  },
  "id": "redis-set-queue-counter",
  "name": "Redis - Set Queue Counter",
  "type": "n8n-nodes-base.redis",
  "typeVersion": 1,
  "position": [650, 0]
}
```
