-- id bigserial PRIMARY KEY
-- workflow_name text NOT NULL
-- workflow_id text NOT NULL
-- execution_id text NOT NULL
-- channel text NOT NULL DEFAULT 'unknown'
-- status text NOT NULL CHECK (status IN ('started', 'completed', 'failed'))
-- started_at timestamptz NOT NULL DEFAULT now()
-- completed_at timestamptz NULL
-- duration_ms integer NULL
-- tenant_id text NULL
-- correlation_id text NULL
-- error_message text NULL
-- metadata_json jsonb NOT NULL DEFAULT '{}'::jsonb
-- created_at timestamptz NOT NULL DEFAULT now()

<!-- ops.workflow_audit_archive: same columns + archived_at timestamptz NOT NULL DEFAULT now() -->

Write endpoint contract (POST body):
{
  "workflow_name": "W1 - IN WhatsApp Adapter",
  "workflow_id": "workflow-uuid",
  "execution_id": "exec-uuid",
  "channel": "whatsapp",
  "status": "started",          // or "completed" or "failed"
  "started_at": "ISO-8601",     // only for status=started
  "completed_at": "ISO-8601",   // only for status=completed or failed
  "duration_ms": 1234,          // only for status=completed or failed
  "tenant_id": "optional",
  "correlation_id": "uuid",
  "error_message": "optional",
  "metadata_json": {}
}

Query endpoint query params:
  start_date: ISO-8601 (inclusive, default: 7 days ago)
  end_date: ISO-8601 (inclusive, default: now)
  workflow_name: string filter (partial match using ILIKE)
  page: integer (default 1)
  page_size: integer (default 50, max 200)

n8n Postgres node: upsert pattern using ON CONFLICT (execution_id, status):
  We cannot use ON CONFLICT (execution_id) alone since we insert twice per execution
  (once for 'started', once for 'completed'/'failed'). Solution: use execution_id + status
  as composite unique key — but simpler: insert for 'started', UPDATE for 'completed'/'failed'
  where execution_id matches AND status = 'started'.
  Actually simplest: two branches in W_AUDIT_WRITE based on status field.
  Branch A (status=started): INSERT
  Branch B (status=completed or failed): UPDATE SET status=..., completed_at=..., duration_ms=...,
    error_message=... WHERE execution_id=$1

Archival batch query (move 500 rows at a time):
  WITH archived AS (
    DELETE FROM ops.workflow_audit
    WHERE id IN (
      SELECT id FROM ops.workflow_audit
      WHERE started_at < now() - interval '90 days'
      LIMIT 500
    )
    RETURNING *
  )
  INSERT INTO ops.workflow_audit_archive
    SELECT *, now() AS archived_at FROM archived;
