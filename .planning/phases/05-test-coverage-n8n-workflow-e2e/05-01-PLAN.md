**Helper: generate_meta_sig()**
```bash
generate_meta_sig() {
  local payload="$1"
  local secret="${META_APP_SECRET}"
  local hex
  hex=$(echo -n "$payload" | openssl dgst -sha256 -hmac "$secret" | sed 's/^.* //')
  echo "sha256=${hex}"
}
```

**Helper: n8n_login()** — Copy the exact login helper from test_harness.sh (try both email and emailOrLdapLoginId formats). Only call if N8N_JAR does not already have valid cookies.

**Helper: poll_db()**
```bash
poll_db() {
  local query="$1"
  local expected="${2:-1}"
  local max_wait="${3:-20}"
  local waited=0
  while [ "${waited}" -lt "${max_wait}" ]; do
    count=$(docker compose -f "${COMPOSE_FILE}" exec -T postgres sh -lc \
      "psql -U n8n -d n8n -Atc \"${query}\"" | tr -d '\r\n')
    if [ "${count:-0}" -ge "${expected}" ]; then
      return 0
    fi
    sleep 1
    waited=$((waited + 1))
  done
  return 1
}
```
