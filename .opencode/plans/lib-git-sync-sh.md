# Library: git-sync.sh

> **Purpose**: Git integration and auto-commit for Obsidient
> **File**: `lib/git-sync.sh`

---

## Full Implementation

```bash
#!/bin/bash
###############################################################################
# Git Sync - Auto-commit Obsidient changes to git
###############################################################################

OBSIDIENT_DIR="${OBSIDIENT_DIR:-.obsidient}"
CONFIG_FILE="$OBSIDIENT_DIR/.obsidientrc"

# Check if git integration is enabled
git_integration_enabled() {
    if [ -f "$CONFIG_FILE" ]; then
        grep -q "auto_commit: true" "$CONFIG_FILE" 2>/dev/null
        return $?
    fi
    return 1
}

# Get git branch from config
git_target_branch() {
    if [ -f "$CONFIG_FILE" ]; then
        grep "branch:" "$CONFIG_FILE" | cut -d' ' -f2
    else
        echo "main"
    fi
}

# Check if we're on the right branch
is_correct_branch() {
    local target_branch=$(git_target_branch)
    local current_branch=$(git branch --show-current 2>/dev/null)
    
    [ "$current_branch" = "$target_branch" ]
}

# Stage Obsidient changes
git_stage_obsidient() {
    if [ -d "$OBSIDIENT_DIR" ]; then
        git add "$OBSIDIENT_DIR/" 2>/dev/null || true
    fi
}

# Commit with message
git_commit_obsidient() {
    local action="$1"
    local target="$2"
    
    # Check if git integration is enabled
    if ! git_integration_enabled; then
        return 0
    fi
    
    # Check if we're in a git repo
    if ! git rev-parse --git-dir > /dev/null 2>&1; then
        echo "Warning: Not in a git repository"
        return 1
    fi
    
    # Check branch
    if ! is_correct_branch; then
        local current=$(git branch --show-current)
        local target=$(git_target_branch)
        echo "Warning: Not on target branch (current: $current, target: $target)"
        return 1
    fi
    
    # Stage changes
    git_stage_obsidient
    
    # Check if there are changes to commit
    if git diff --cached --quiet; then
        return 0
    fi
    
    # Generate commit message
    local timestamp=$(date -u +"%Y-%m-%dT%H:%M:%SZ")
    local commit_msg="[obsidient] $action: $target

Auto-generated by Ralphé Unified System
Timestamp: $timestamp"
    
    # Commit
    git commit -m "$commit_msg" 2>/dev/null || {
        echo "Warning: Git commit failed"
        return 1
    }
    
    echo "Committed: $action $target"
}

# Push to remote (optional)
git_push_obsidient() {
    if git_integration_enabled; then
        local branch=$(git_target_branch)
        git push origin "$branch" 2>/dev/null || {
            echo "Warning: Git push failed"
            return 1
        }
    fi
}

# Get last commit info for Obsidient
git_last_commit_info() {
    git log -1 --format="%h|%s|%ci" 2>/dev/null || echo "none|no commits|"
}

# Sync git status to Obsidient context
sync_git_to_context() {
    local context_file="$OBSIDIENT_DIR/context/current-session.md"
    
    if [ ! -f "$context_file" ]; then
        return 0
    fi
    
    local commit_info=$(git_last_commit_info)
    local commit_hash=$(echo "$commit_info" | cut -d'|' -f1)
    local commit_msg=$(echo "$commit_info" | cut -d'|' -f2)
    local commit_date=$(echo "$commit_info" | cut -d'|' -f3)
    
    # Update context file
    if grep -q "## Git Status" "$context_file"; then
        # Update existing section
        sed -i "/## Git Status/,/## /{s/Last Commit: .*/Last Commit: $commit_hash ($commit_date)/}" "$context_file"
    else
        # Add new section
        sed -i "/^---/,/---/{/---/a\\
## Git Status\nLast Commit: $commit_hash ($commit_date)\nMessage: $commit_msg\n
}" "$context_file"
    fi
}

# Create .gitignore entries for Obsidient
git_setup_obsidient_ignore() {
    local gitignore=".gitignore"
    
    # Check if already configured
    if [ -f "$gitignore" ]; then
        if grep -q ".obsidient/context/snapshots" "$gitignore"; then
            return 0
        fi
    fi
    
    # Add Obsidient entries
    cat >> "$gitignore" << 'EOF'

# Obsidient - Ignore snapshots but keep structure
.obsidient/context/snapshots/*
!.obsidient/context/snapshots/.gitkeep
.obsidient/.sequence
EOF

    echo "Updated .gitignore for Obsidient"
}

# Setup git hooks for Obsidient
git_setup_hooks() {
    local hooks_dir=".git/hooks"
    
    if [ ! -d "$hooks_dir" ]; then
        return 1
    fi
    
    # Post-commit hook to update dashboard
    cat > "$hooks_dir/post-commit-obsidient" << 'EOF'
#!/bin/bash
# Update Obsidient dashboard after commit
if [ -f "lib/obsidient-lib.sh" ]; then
    source lib/obsidient-lib.sh
    update_dashboard
fi
EOF

    chmod +x "$hooks_dir/post-commit-obsidient"
    
    echo "Git hooks configured"
}

# Full setup
git_setup_full() {
    echo "Setting up Git integration for Obsidient..."
    
    git_setup_obsidient_ignore
    git_setup_hooks
    
    echo "Git integration configured"
    echo "Auto-commit: $(git_integration_enabled && echo "enabled" || echo "disabled")"
    echo "Target branch: $(git_target_branch)"
}
```

## Commit Message Format

```
[obsidient] <action>: <target>

Auto-generated by Ralphé Unified System
Timestamp: 2026-04-04T14:45:00Z
```

### Actions
- `task-create`: New task created
- `task-update`: Task updated
- `task-switch`: Active task changed
- `task-archive`: Task archived
- `decision`: Decision recorded
- `context-sync`: Context updated

## Usage Example

```bash
# Source libraries
source lib/obsidient-lib.sh
source lib/git-sync.sh

# Setup git integration
git_setup_full

# After making changes
git_commit_obsidient "task-create" "TASK-001"

# Sync git status to context
sync_git_to_context

# Optional: Push to remote
git_push_obsidient
```
